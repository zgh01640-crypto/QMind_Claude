#### 编码处理逻辑

```python
def strip_serial(item_code: str) -> str:
    """去掉清单编码末尾3位流水号，返回9位基准编码。
    例如: 010102002004 → 010102002
    """
    code = item_code.strip().replace(' ', '')
    if len(code) >= 3:
        return code[:-3]  # 去掉末尾3位
    return code
```

#### 数据库查询

```python
def exec_check_item_code(conn, item_code: str, item_name: str) -> dict:
    """执行工具：查询 tqdk_tqdzm 表"""
    base_code = strip_serial(item_code)  # 010102002004 → 010102002
    
    with conn.cursor() as cur:
        # 查询表: tqdk_tqdzm
        # 字段: zmbh (编码), zmmc (名称)
        cur.execute(
            "SELECT zmmc FROM tqdk_tqdzm WHERE zmbh = %s LIMIT 5",
            (base_code,)
        )
        rows = cur.fetchall()
    
    standard_names = list({r[0] for r in rows if r[0]})
    
    return {
        "item_code": item_code,          # 原始12位编码
        "base_code": base_code,          # 去掉末尾3位
        "item_name": item_name,          # 工程清单名称
        "standard_names": standard_names,  # 标准库匹配的名称列表
        "found": len(standard_names) > 0   # 是否找到
    }
```

### 代码流程

**SSE 端点**: `api/routers/bs2024_match.py:790-829`

```python
@router.post("/bs2024-match/match-item-stream")
def bs2024_match_item_stream(req: SingleMatchRequest):
    """单条清单项 Step 1：编码核查（SSE 流式）"""
    
    def generate():
        # 1️⃣ 读清单项
        row = cur.execute(
            "SELECT id,item_code,item_name,item_description,unit,quantity FROM boq_items WHERE id=%s",
            (req.boq_item_id,)
        ).fetchone()
        
        boq_item = {
            "id": row[0],
            "item_code": row[1],
            "item_name": row[2],
            ...
        }
        
        # 2️⃣ 构建系统提示词
        chapter_name, sp = build_bs2024_system_prompt(conn, req.chapter_ids)
        
        # 3️⃣ 发送 item_info 事件
        yield f"data: {json.dumps({'type':'item_info', 'item':boq_item, ...})}\n\n"
        
        # 4️⃣ 调用 Round 1 流式推理
        for event_type, data in stream_match_bs2024_item_step1(boq_item, sp, conn):
            if event_type == "reasoning_token":
                yield f"data: {json.dumps({'type':'reasoning_token','token':data})}\n\n"
            elif event_type == "code_check":
                yield f"data: {json.dumps({'type':'code_check',**data})}\n\n"
    
    return StreamingResponse(generate(), media_type="text/event-stream")
```
